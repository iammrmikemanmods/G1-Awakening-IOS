    enable_language(OBJCXX)
    add_library(g1_metal STATIC platform/apple/MetalRenderer.mm)
    target_include_directories(g1_metal PUBLIC platform/apple)
    target_link_libraries(g1_metal PUBLIC g1_resources "-framework Foundation" "-framework Metal")
    target_compile_options(g1_metal PRIVATE -fobjc-arc)
    set_target_properties(g1_metal PROPERTIES OBJCXX_STANDARD 17 OBJCXX_STANDARD_REQUIRED YES)
    if(CMAKE_SYSTEM_NAME STREQUAL "iOS" AND G1_BUILD_APP)
        set(G1_SOURCE_COMMIT "unknown" CACHE STRING "Source commit recorded in the test app")
