cmake_minimum_required(VERSION 3.16)
project(G1AwakeningResources LANGUAGES CXX)
find_package(ZLIB REQUIRED)
add_library(g1_resources STATIC src/Archive.cpp src/M3G.cpp src/Mission.cpp src/Movement.cpp src/Atlas.cpp)
target_include_directories(g1_resources PUBLIC include)
target_compile_features(g1_resources PUBLIC cxx_std_17)
target_link_libraries(g1_resources PUBLIC ZLIB::ZLIB)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
    include(CTest)
    add_executable(g1_asset_probe tools/g1_asset_probe.cpp)
    target_link_libraries(g1_asset_probe PRIVATE g1_resources)
    add_executable(m3g_probe tools/m3g_probe.cpp)
    target_link_libraries(m3g_probe PRIVATE g1_resources)
    add_executable(mission_probe tools/mission_probe.cpp)
    target_link_libraries(mission_probe PRIVATE g1_resources)
    add_executable(atlas_probe tools/atlas_probe.cpp)
    target_link_libraries(atlas_probe PRIVATE g1_resources)
    if(BUILD_TESTING)
        add_executable(g1_mission_tests tests/mission_test.cpp)
        target_link_libraries(g1_mission_tests PRIVATE g1_resources)
        add_test(NAME mission_format_contract COMMAND g1_mission_tests)
        add_executable(g1_movement_tests tests/movement_test.cpp)
        target_link_libraries(g1_movement_tests PRIVATE g1_resources)
        add_test(NAME movement_original_function_vectors COMMAND g1_movement_tests
                 "${CMAKE_CURRENT_SOURCE_DIR}/tests/movement_cases.txt")
        add_executable(g1_atlas_tests tests/atlas_test.cpp)
        target_link_libraries(g1_atlas_tests PRIVATE g1_resources)
        add_test(NAME atlas_format_geometry_contract COMMAND g1_atlas_tests)
    endif()
endif()
